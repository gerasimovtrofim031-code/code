interface Controllable {
    void turnOn();
    void turnOff();
    boolean isOn();
}

abstract class Device implements Controllable {
    protected String id;
    protected String name;
    protected boolean status;
    protected double powerConsumption;

    public Device(String id, String name, boolean status, double powerConsumption) {
        this.id = id;
        this.name = name;
        this.status = status;
        this.powerConsumption = powerConsumption;
    }

    @Override
    public void turnOn() {
        this.status = true;
    }

    @Override
    public void turnOff() {
        this.status = false;
    }

    @Override
    public boolean isOn() {
        return this.status;
    }

    public abstract String getDetails();

    public String getId() { return id; }
    public void setId(String id) { this.id = id; }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }

    public boolean isStatus() { return status; }
    public void setStatus(boolean status) { this.status = status; }

    public double getPowerConsumption() { return powerConsumption; }
    public void setPowerConsumption(double powerConsumption) { this.powerConsumption = powerConsumption; }

    @Override
    public String toString() {
        return getDetails();
    }
}

class SmartLight extends Device {
    private int brightness; // 0 - 100
    private String color;

    public SmartLight(String id, String name, boolean status, double powerConsumption, int brightness, String color) {
        super(id, name, status, powerConsumption);
        this.brightness = Math.max(0, Math.min(100, brightness));
        this.color = color;
    }

    public int getBrightness() { return brightness; }
    public void setBrightness(int brightness) {
        this.brightness = Math.max(0, Math.min(100, brightness));
    }

    public String getColor() { return color; }
    public void setColor(String color) { this.color = color; }

    @Override
    public String getDetails() {
        return String.format("[Умная лампа] ID: %s | Имя: '%s' | Статус: %s | Потребление: %.1f Вт | Яркость: %d%% | Цвет: %s",
                id, name, (status ? "ВКЛ" : "ВЫКЛ"), powerConsumption, brightness, color);
    }
}

class SmartThermostat extends Device {
    private double targetTemperature;

    public SmartThermostat(String id, String name, boolean status, double powerConsumption, double targetTemperature) {
        super(id, name, status, powerConsumption);
        this.targetTemperature = targetTemperature;
    }

    public double getTargetTemperature() { return targetTemperature; }
    public void setTargetTemperature(double targetTemperature) { this.targetTemperature = targetTemperature; }

    @Override
    public String getDetails() {
        return String.format("[Термостат] ID: %s | Имя: '%s' | Статус: %s | Потребление: %.1f Вт | Целевая темп.: %.1f°C",
                id, name, (status ? "ВКЛ" : "ВЫКЛ"), powerConsumption, targetTemperature);
    }
}

class SmartSocket extends Device {
    private double currentLoad;

    public SmartSocket(String id, String name, boolean status, double powerConsumption, double currentLoad) {
        super(id, name, status, powerConsumption);
        this.currentLoad = currentLoad;
    }

    public double getCurrentLoad() { return currentLoad; }
    public void setCurrentLoad(double currentLoad) { this.currentLoad = currentLoad; }

    @Override
    public String getDetails() {
        return String.format("[Умная розетка] ID: %s | Имя: '%s' | Статус: %s | Потребление: %.1f Вт | Нагрузка: %.1f Вт",
                id, name, (status ? "ВКЛ" : "ВЫКЛ"), powerConsumption, currentLoad);
    }
}

class Scenario<T extends Device> {
    private final String name;
    private final Predicate<T> condition;
    private final Consumer<T> action;

    public Scenario(String name, Predicate<T> condition, Consumer<T> action) {
        this.name = name;
        this.condition = condition;
        this.action = action;
    }

    public String getName() {
        return name;
    }

    public void apply(T device) {
        if (condition.test(device)) {
            action.accept(device);
        }
    }
}

class SmartHomeManager {
    private final Map<String, List<Device>> rooms = new LinkedHashMap<>();
    private final List<Scenario<? extends Device>> scenarios = new ArrayList<>();

    public Map<String, List<Device>> getRooms() {
        return rooms;
    }

    public List<Scenario<? extends Device>> getScenarios() {
        return scenarios;
    }

    public void addDevice(String room, Device device) {
        rooms.computeIfAbsent(room, k -> new ArrayList<>()).add(device);
    }

    public void addScenario(Scenario<?> scenario) {
        scenarios.add(scenario);
    }

    public Device getDeviceById(String id) {
        for (List<Device> deviceList : rooms.values()) {
            for (Device device : deviceList) {
                if (device.getId().equalsIgnoreCase(id.trim())) {
                    return device;
                }
            }
        }
        return null;
    }

    public void executeAllScenarios() {
        if (scenarios.isEmpty()) {
            System.out.println("(!) В системе нет добавленных сценариев. Добавьте базовый (3) или индивидуальный (4) сценарий.");
            return;
        }

        System.out.println("\n--- Запуск проверки всех сценариев автоматизации ---");
        for (Map.Entry<String, List<Device>> entry : rooms.entrySet()) {
            for (Device device : entry.getValue()) {
                for (Scenario<? extends Device> scenario : scenarios) {
                    tryApplyScenario(scenario, device);
                }
            }
        }
        System.out.println("--- Проверка сценариев завершена ---\n");
    }

    @SuppressWarnings("unchecked")
    private <T extends Device> void tryApplyScenario(Scenario<T> scenario, Device device) {
        try {
            scenario.apply((T) device);
        } catch (ClassCastException ignored) {
            // Устройство несовместимого типа пропускается сценарием
        }
    }

    public Stream<Device> getAnalyticsStream() {
        return rooms.values().stream().flatMap(List::stream);
    }
}

            case 36: return new VariantConfig(36, ScenarioType.CALM_SLEEP, AnalyticsType.TOTAL_POWER);
            default: throw new IllegalArgumentException("Номер варианта должен быть в диапазоне от 1 до 36!");
        }
    }
}

    private static void printMenu(VariantConfig variant) {

    private static void handleAddNewDevice(SmartHomeManager manager, Scanner scanner) {
        System.out.println("\n--- Добавление нового устройства ---");
        System.out.print("Введите название комнаты: ");
        String room = scanner.nextLine().trim();

        System.out.println("Выберите тип устройства:");
        System.out.println(" 1. SmartLight (Умная лампа)");
        System.out.println(" 2. SmartThermostat (Умный термостат)");
        System.out.println(" 3. SmartSocket (Умная розетка)");
        System.out.print("Ваш выбор: ");
        String type = scanner.nextLine().trim();

        System.out.print("Введите уникальный ID: ");
        String id = scanner.nextLine().trim();
        if (manager.getDeviceById(id) != null) {
            System.out.println("(!) Устройство с таким ID уже существует!");
            return;
        }

        System.out.print("Введите имя устройства (например, 'Люстра Коридор', 'TV Зал'): ");
        String name = scanner.nextLine().trim();

        System.out.print("Устройство включено? (1 - Да, 0 - Нет): ");
        boolean status = "1".equals(scanner.nextLine().trim());

        System.out.print("Текущее энергопотребление (Вт, например 45.0): ");
        double power = parseDouble(scanner.nextLine().trim(), 0.0);

        Device createdDevice;
        switch (type) {
            case "1":
                System.out.print("Яркость (0 - 100): ");
                int brightness = parseInt(scanner.nextLine().trim(), 50);
                System.out.print("Цвет свечения (например, White, Warm, Red): ");
                String color = scanner.nextLine().trim();
                createdDevice = new SmartLight(id, name, status, power, brightness, color);
                break;
            case "2":
                System.out.print("Целевая температура (°C, например 22.5): ");
                double temp = parseDouble(scanner.nextLine().trim(), 20.0);
                createdDevice = new SmartThermostat(id, name, status, power, temp);
                break;
            case "3":
                System.out.print("Подключенная нагрузка (Вт, например 2100.0): ");
                double load = parseDouble(scanner.nextLine().trim(), power);
                createdDevice = new SmartSocket(id, name, status, power, load);
                break;
            default:
                System.out.println("(!) Неверный тип устройства. Отмена добавления.");
                return;
        }

        manager.addDevice(room, createdDevice);
        System.out.println("-> Устройство успешно добавлено: " + createdDevice.getDetails());
    }

    private static void handleToggleDevice(SmartHomeManager manager, Scanner scanner) {
        System.out.print("Введите ID устройства для изменения статуса: ");
        String id = scanner.nextLine().trim();
        Device device = manager.getDeviceById(id);

        if (device == null) {
            System.out.println("(!) Устройство с ID '" + id + "' не найдено.");
            return;
        }

        if (device.isOn()) {
            device.turnOff();
            System.out.printf("-> Устройство '%s' (ID: %s) успешно ВЫКЛЮЧЕНО.%n", device.getName(), device.getId());
        } else {
            device.turnOn();
            System.out.printf("-> Устройство '%s' (ID: %s) успешно ВКЛЮЧЕНО.%n", device.getName(), device.getId());
        }
    }

    private static void handleAddBaseScenario(SmartHomeManager manager) {
        Scenario<Device> phantomScenario = new Scenario<>(
                "Сброс фантомного потребления",
                d -> !d.isOn() && d.getPowerConsumption() > 0.0,
                d -> {
                    System.out.printf("[Сработал базовый сценарий] Устройство '%s' выключено, но потребляло %.1f Вт. Сброшено в 0.0 Вт.%n",
                            d.getName(), d.getPowerConsumption());
                    d.setPowerConsumption(0.0);
                }
        );
        manager.addScenario(phantomScenario);
        System.out.println("-> Базовый сценарий «Сброс фантомного потребления» добавлен в систему.");
    }

    private static void handleAddVariantScenario(SmartHomeManager manager, VariantConfig variant) {
        Scenario<? extends Device> sc = variant.getScenarioType().create();
        manager.addScenario(sc);
        System.out.printf("-> Индивидуальный сценарий варианта № %d «%s» успешно добавлен в систему.%n",
                variant.getVariantNumber(), sc.getName());
    }

    private static void handlePrintAllDevices(SmartHomeManager manager) {
        System.out.println("\n================ ВСЕ УСТРОЙСТВА В СИСТЕМЕ ================");
        if (manager.getRooms().isEmpty()) {
            System.out.println("В доме пока нет ни одной комнаты и ни одного добавленного устройства.");
        } else {
            manager.getRooms().forEach((room, devices) -> {
                System.out.printf("%nКомната: [%s] (Количество устройств: %d)%n", room, devices.size());
                if (devices.isEmpty()) {
                    System.out.println("  (в этой комнате нет устройств)");
                } else {
                    for (int i = 0; i < devices.size(); i++) {
                        System.out.printf("  %d. %s%n", i + 1, devices.get(i).getDetails());
                    }
                }
            });
        }
        System.out.println("==========================================================\n");
    }

    private static double parseDouble(String text, double defaultValue) {
        try {
            return Double.parseDouble(text.replace(',', '.'));
        } catch (Exception e) {
            return defaultValue;
        }
    }

    private static int parseInt(String text, int defaultValue) {
        try {
            return Integer.parseInt(text);
        } catch (Exception e) {
            return defaultValue;
        }
    }
}
